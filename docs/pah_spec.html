396    def _calc_c_abs(self, wavelength_arr, radius_arr):
397        """Internals for calc_c_abs(). Note that Astropy Units have been stripped for performance reasons.
398
399        Parameters
400        ----------
401        wavelength_arr : numpy.ndarray
402            Array of wavelengths to calculate C_abs for (in um)
403        radius_arr : numpy.ndarray
404            Array of dust grain radii to calculate C_abs for (in Angstroms)
405
406        Returns
407        -------
408        c_abs_ion_out : numpy.ndarray
409            Array with C_abs values for ionized grains (in cm**2)
410        c_abs_neu_out : numpy.ndarray
411            Array with C_abs values for neutral grains (in cm**2)
412
413        Raises
414        ------
415        AttributeError
416            If the input is not an astropy.units.Quantity object
417        TypeError
418            If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
419        """
420
421        nc = _calc_nc(radius_arr * u.AA)
422        hc = 0.5 * np.ones_like(nc)  # Note that this is from Equation 4 of DL07
423        hc[(nc >= 25) & (nc <= 100)] = 0.5 * np.sqrt(25 / nc[(nc >= 25) & (nc <= 100)])
424        hc[nc > 100] = 0.25
425        xi_gra = np.zeros_like(radius_arr)
426        xi_gra[radius_arr < 50] = 0.01
427        xi_gra[radius_arr >= 50] = 0.01 + 0.99 * (1.0 - (50 / radius_arr[radius_arr >= 50]) ** 3)
428
429        def s_func(lam, lamj, gamma, sigma):
430            num = 2 * gamma * lamj * sigma
431            denom = np.pi * (((lam / lamj) - (lamj / lam)) ** 2 + gamma**2)
432            return num / denom
433
434        def c_func(y):
435            return np.arctan(1.0e3 * (y - 1.0) ** 3 / y) / np.pi + 0.5
436
437        # Cutoff function parameters
438        m_ring = 0.3 * nc
439        m_ring[nc > 40] = 0.4 * nc[nc > 40]
440        lamc_neu = 0.951 / (1.0 + 3.616 / np.sqrt(m_ring))  # um
441        lamc_ion = 1.125 / (1.0 + 2.567 / np.sqrt(m_ring))  # um
442
443        x = 1.0 / wavelength_arr  # um
444        c_abs_ion_out = np.zeros((len(radius_arr), len(wavelength_arr)))  # cm**2
445        c_abs_neu_out = np.zeros((len(radius_arr), len(wavelength_arr)))  # cm**2
446
447        for i in range(len(radius_arr)):
448            # cabs_graphite_spl expects units of um for both radius and wavelength parameters
449            graph_cont = self._cabs_graphite_spl.ev(radius_arr[i] * 0.0001, wavelength_arr)  # cm**2
450            c_abs_ion_out[i, :] += xi_gra[i] * graph_cont  # cm**2
451            c_abs_neu_out[i, :] += xi_gra[i] * graph_cont  # cm**2
452
453            # PAH contribution
454            c_fac_neu = c_func(
455                (wavelength_arr / lamc_neu[i]) ** -1
456            )  # Note error in Equation A7, should be C(lambda_c/lambda)
457            c_fac_ion = c_func(
458                (wavelength_arr / lamc_ion[i]) ** -1
459            )  # Note error in Equation A7, should be C(lambda_c/lambda)
460            s_mat_neu = np.zeros((len(self._lamj_tab), len(wavelength_arr)))  # cm**2
461            s_mat_ion = np.zeros((len(self._lamj_tab), len(wavelength_arr)))  # cm**2
462            for j in range(len(self._lamj_tab)):
463                if self._hc_tab[j] == 0:
464                    # s_func param units: um -> cm, um -> cm, dimensionless, cm; units of s_func are cm**2
465                    s_mat_neu[j, :] = s_func(
466                        wavelength_arr * 1e-4, self._lamj_tab[j] * 1e-4, self._gamj_tab[j], self._sigj_neu_tab[j]
467                    )
468                    s_mat_ion[j, :] = s_func(
469                        wavelength_arr * 1e-4, self._lamj_tab[j] * 1e-4, self._gamj_tab[j], self._sigj_ion_tab[j]
470                    )
471                else:
472                    s_mat_neu[j, :] = s_func(
473                        wavelength_arr * 1e-4,
474                        self._lamj_tab[j] * 1e-4,
475                        self._gamj_tab[j],
476                        self._sigj_neu_tab[j] * hc[i],
477                    )
478                    s_mat_ion[j, :] = s_func(
479                        wavelength_arr * 1e-4,
480                        self._lamj_tab[j] * 1e-4,
481                        self._gamj_tab[j],
482                        self._sigj_ion_tab[j] * hc[i],
483                    )
484
485            idx = np.where((10 < x) & (x < 15))
486            c_abs_ion_out[i, idx] += (s_mat_ion[0, idx] + (1.35 * x[idx] - 3.0) * 1.0e-18) * nc[i] * (1.0 - xi_gra[i])
487            c_abs_neu_out[i, idx] += (s_mat_neu[0, idx] + (1.35 * x[idx] - 3.0) * 1.0e-18) * nc[i] * (1.0 - xi_gra[i])
488
489            idx = np.where((7.7 < x) & (x <= 10))
490            c_abs_ion_out[i, idx] += (
491                (66.302 - 24.367 * x[idx] + 2.950 * x[idx] ** 2 - 0.1057 * x[idx] ** 3)
492                * 1.0e-18
493                * nc[i]
494                * (1.0 - xi_gra[i])
495            )
496            c_abs_neu_out[i, idx] += (
497                (66.302 - 24.367 * x[idx] + 2.950 * x[idx] ** 2 - 0.1057 * x[idx] ** 3)
498                * 1.0e-18
499                * nc[i]
500                * (1.0 - xi_gra[i])
501            )
502
503            idx = np.where((5.9 < x) & (x <= 7.7))
504            c0 = 1.8687e-18  # cm**2
505            c1 = 1.905e-19  # cm**2
506            c2 = 4.175e-19  # cm**2
507            c3 = 4.37e-20  # cm**2, Note Equations A11 and A12 are mislabeled
508            c_abs_ion_out[i, idx] += (
509                (s_mat_ion[1, idx] + c0 + c1 * x[idx] + c2 * (x[idx] - 5.9) ** 2 + c3 * (x[idx] - 5.9) ** 3)
510                * nc[i]
511                * (1.0 - xi_gra[i])
512            )
513            c_abs_neu_out[i, idx] += (
514                (s_mat_neu[1, idx] + c0 + c1 * x[idx] + c2 * (x[idx] - 5.9) ** 2 + c3 * (x[idx] - 5.9) ** 3)
515                * nc[i]
516                * (1.0 - xi_gra[i])
517            )
518
519            idx = np.where((3.3 < x) & (x <= 5.9))
520            c_abs_ion_out[i, idx] += (s_mat_ion[1, idx] + c0 + c1 * x[idx]) * nc[i] * (1.0 - xi_gra[i])
521            c_abs_neu_out[i, idx] += (s_mat_neu[1, idx] + c0 + c1 * x[idx]) * nc[i] * (1.0 - xi_gra[i])
522
523            idx = np.where(x <= 3.3)
524            c_abs_ion_out[i, idx] += 34.58e-18 * 10 ** (-3.431 / x[idx]) * c_fac_ion[idx] * nc[i] * (1.0 - xi_gra[i])
525            c_abs_neu_out[i, idx] += 34.58e-18 * 10 ** (-3.431 / x[idx]) * c_fac_neu[idx] * nc[i] * (1.0 - xi_gra[i])
526
527            for j in range(len(self._lamj_tab)):
528                if j > 1:
529                    c_abs_ion_out[i, idx] += s_mat_ion[j, idx] * nc[i] * (1.0 - xi_gra[i])
530                    c_abs_neu_out[i, idx] += s_mat_neu[j, idx] * nc[i] * (1.0 - xi_gra[i])
531
532            c_abs_ion_out[i, idx] += (
533                3.5e-19 * 10 ** (-1.45 / x[idx]) * np.exp(-((0.1 * x[idx]) ** 2)) * nc[i] * (1.0 - xi_gra[i])
534            )  # Note: does not appear in D21+
535
536        return c_abs_ion_out, c_abs_neu_out

538    def _scale_basis_spectra(
539        self,
540        photon_wavelength_arr,
541        dlambda,
542        wavelength_arr,
543        grain_radius,
544        wavelength_arr_u,
545        u_lambda_arr,
546        p_lambda_arr,
547        ion,
548    ):
549        """Calculate the energy conservation normalization to scale basis vectors to the input radiation field.
550        Note that Astropy Units attributes should be stripped from input parameters (for performance reasons).
551
552        Parameters
553        ----------
554        photon_wavelength_arr : numpy.ndarray
555            Array of absorbed photon wavelengths (in microns)
556        dlambda : float
557            Width of the "monochromatic" radiation field, defined as a percentage of lambda_abs
558        wavelength_arr : numpy.ndarray
559            Array of emission wavelengths (in microns)
560        grain_radius : float
561            Dust grain radius (in Angstroms)
562        wavelength_arr_u : numpy.ndarray
563            Wavelength array for the radiation field u_lambda (in microns)
564        u_lambda_arr : numpy.ndarray
565            Array of length len(wavelengths_u) with the radiation field (in erg / cm**4)
566        p_lambda_arr : numpy.ndarray
567            Basis vector array of size (len(photon_wavelength_arr), len(wavelength_arr)) for a given grain
568            (in erg / (cm * s))
569        ion : bool
570            Specifies whether or not the PAH is ionized, used to calculate the cross-section
571
572        Returns
573        -------
574        scaled_spectrum : array_like
575            Integrated spectrum for the input grain and radiation field (in erg / (cm * s)) with length
576            len(wavelength_arr))
577        """
578
579        # call calc_c_abs once for all wavelength values needed by calc_normalization
580        lambda1_max = photon_wavelength_arr[-1] + photon_wavelength_arr[-1] * dlambda
581        if ion:
582            c_abs_arr_u = self._calc_c_abs(wavelength_arr_u, np.array([grain_radius]))[0][0]
583            c_abs_phot_arr = np.concatenate(
584                [
585                    self._calc_c_abs(photon_wavelength_arr, np.array([grain_radius]))[0][0],
586                    self._calc_c_abs(np.array([lambda1_max]), np.array([grain_radius]))[0][0],
587                ],
588            )
589        else:
590            c_abs_arr_u = self._calc_c_abs(wavelength_arr_u, np.array([grain_radius]))[1][0]
591            c_abs_phot_arr = np.concatenate(
592                [
593                    self._calc_c_abs(photon_wavelength_arr, np.array([grain_radius]))[1][0],
594                    self._calc_c_abs(np.array([lambda1_max]), np.array([grain_radius]))[1][0],
595                ],
596            )
597
598        normalizations = np.zeros(len(photon_wavelength_arr))
599
600        for i, lambda_abs in enumerate(photon_wavelength_arr):
601            # define wavelengths of monochromatic radiation field (MRF)
602            lambda0, lambda1 = lambda_abs, lambda_abs + lambda_abs * dlambda
603            wh_mrf = np.where(np.logical_and(wavelength_arr_u > lambda0, wavelength_arr_u < lambda1))
604            wavelength_arr_mrf = np.concatenate([[lambda0], wavelength_arr_u[wh_mrf], [lambda1]])
605

607            wh0 = np.argmin(np.abs(lambda0 - photon_wavelength_arr))
608            wh1 = np.argmin(np.abs(lambda1 - photon_wavelength_arr))
609            c_abs_arr_mrf = np.concatenate([[c_abs_phot_arr[wh0]], c_abs_arr_u[wh_mrf], [c_abs_phot_arr[wh1]]])
610
611            # interpolate radiation field to the MRF wavelength range
612            u_lambda_arr_mrf = np.interp(wavelength_arr_mrf, wavelength_arr_u, u_lambda_arr)
613
614            # integrate to determine the power of the radiation field in this wavelength range
615            numerator = trapezoid(
616                u_lambda_arr_mrf * c_abs_arr_mrf * _C_CGS,
617                x=wavelength_arr_mrf * 1e-4,  # convert wavelength_arr_mrf from u.um to u.cm
618            )
619
620            # integrate to determine the power radiated by the grain over all wavelengths
621            denominator = trapezoid(
622                p_lambda_arr[i], x=wavelength_arr * 1e-4  # convert wavelength_arr from u.um to u.cm
623            )
624
625            normalizations[i] = numerator / denominator
626
627        return p_lambda_arr * normalizations[:, np.newaxis]
