 147    def generate_spectrum(self, wavelength_arr=None, u_lambda_arr=None, size_dist_neu=None, size_dist_ion=None):
 148        """Scale the basis spectra for ionized and neutral PAHs to an input radiation field.
 149
 150        Parameters
 151        ----------
 152        wavelength_arr : astropy.units.Quantity (array_like), optional
 153            Wavelength array for the radiation field u_lambda (in u.um)
 154        u_lambda_arr : astropy.units.Quantity (array_like), optional
 155            Array of length len(wavelength_arr) with the radiation field (in u.erg / u.cm ** 4)
 156        size_dist_neu : array_like, optional
 157            PAH0 size distribution and ionization function for each grain size in pah_spec.GRAIN_SIZES
 158        size_dist_ion : array_like, optional
 159            PAH+ size distribution and ionization function for each grain size in pah_spec.GRAIN_SIZES
 160
 161        Returns
 162        -------
 163        spectrum_neu : astropy.units.Quantity (array_like)
 164            PAH0 size- and ionization-integrated spectrum for grains heated by the input u_lambda
 165            (in u.erg / (u.cm * u.s))
 166        spectrum_ion : astropy.units.Quantity (array_like)
 167            PAH+ size- and ionization-integrated spectrum for grains heated by the input u_lambda
 168            (in u.erg / (u.cm * u.s))
 169
 170        Raises
 171        ------
 172        AttributeError
 173            If the input is not an astropy.units.Quantity object
 174        TypeError
 175            If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 176        """
 177        # If the radiation field, size distribution, or ionization function is not specified, then use the defaults
 178        if wavelength_arr is None:
 179            wavelength_arr = self.wavelength_u_arr
 180        if u_lambda_arr is None:
 181            u_lambda_arr = self.u_lambda_arr
 182        if size_dist_neu is None:
 183            size_dist_neu = self.size_dist_neu
 184        if size_dist_ion is None:
 185            size_dist_ion = self.size_dist_ion
 186
 187        _check_param(wavelength_arr, u.um)
 188        _check_param(u_lambda_arr, u.erg / u.cm**4)
 189
 190        spectra_neu_a = np.zeros((len(GRAIN_SIZES), len(self.emission_wavelengths))) * u.erg / (u.cm * u.s)
 191        spectra_ion_a = np.zeros((len(GRAIN_SIZES), len(self.emission_wavelengths))) * u.erg / (u.cm * u.s)
 192        for i, grain_radius in enumerate(GRAIN_SIZES):
 193            spectra_neu_a[i] = np.sum(
 194                self._scale_basis_spectra(
 195                    self.photon_wavelengths,
 196                    _DELTA_LAMBDA,
 197                    self.emission_wavelengths,
 198                    grain_radius.to(u.AA),
 199                    wavelength_arr,
 200                    u_lambda_arr,
 201                    self.basis_spectra_neu[i],
 202                    ion=False,
 203                ),
 204                axis=0,
 205            )
 206            spectra_ion_a[i] = np.sum(
 207                self._scale_basis_spectra(
 208                    self.photon_wavelengths,
 209                    _DELTA_LAMBDA,
 210                    self.emission_wavelengths,
 211                    grain_radius.to(u.AA),
 212                    wavelength_arr,
 213                    u_lambda_arr,
 214                    self.basis_spectra_ion[i],
 215                    ion=True,
 216                ),
 217                axis=0,
 218            )
 219
 220        spectrum_neu = _size_integrate(spectra_neu_a, size_dist_neu)
 221        spectrum_ion = _size_integrate(spectra_ion_a, size_dist_ion)
 222
 223        return spectrum_neu, spectrum_ion

 328    def calc_c_abs(self, wavelength_arr, radius_arr):
 329        """Calculate the absorption cross-section, C_abs, for input grain sizes and wavelengths based on method from
 330        Draine et al. (2021).
 331
 332        Parameters
 333        ----------
 334        wavelength_arr : astropy.units.Quantity (array_like)
 335            Array of wavelengths to calculate C_abs for
 336        radius_arr : astropy.units.Quantity (float or array_like)
 337            Array of dust grain radii to calculate C_abs for
 338
 339        Returns
 340        -------
 341        c_abs_ion_out : astropy.units.Quantity (array_like)
 342            Array with C_abs values for ionized grains (in u.cm ** 2)
 343        c_abs_neu_out : astropy.units.Quantity (array_like)
 344            Array with C_abs values for neutral grains (in u.cm ** 2)
 345
 346        Raises
 347        ------
 348        AttributeError
 349            If the input is not an astropy.units.Quantity object
 350        TypeError
 351            If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 352        """
 353        # TODO: check that grain does not exceed maximum allowed size
 354
 355        wavelength_unit, radius_unit = u.um, u.AA
 356
 357        # ensure that wavelength and grain radius variables are array-like with correct units
 358        wavelength_arr = _check_param(wavelength_arr, wavelength_unit, force_iterable=True)
 359        radius_arr = _check_param(radius_arr, radius_unit, force_iterable=True)
 360
 361        nc = _calc_nc(radius_arr)
 362        hc = 0.5 * np.ones_like(nc)  # Note that this is from Equation 4 of DL07
 363        hc[(nc >= 25) & (nc <= 100)] = 0.5 * np.sqrt(25 / nc[(nc >= 25) & (nc <= 100)])
 364        hc[nc > 100] = 0.25
 365        xi_gra = np.zeros_like(radius_arr.value)
 366        xi_gra[radius_arr < 50 * u.AA] = 0.01
 367        xi_gra[radius_arr >= 50 * u.AA] = 0.01 + 0.99 * (1.0 - (50 * u.AA / radius_arr[radius_arr >= 50 * u.AA]) ** 3)
 368
 369        def s_func(lam, lamj, gamma, sigma):
 370            num = 2 * gamma * lamj * sigma
 371            denom = np.pi * (((lam / lamj) - (lamj / lam)) ** 2 + gamma**2)
 372            return num / denom
 373
 374        def c_func(y):
 375            return np.arctan(1.0e3 * (y - 1.0) ** 3 / y) / np.pi + 0.5
 376
 377        # Cutoff function parameters
 378        m_ring = 0.3 * nc
 379        m_ring[nc > 40] = 0.4 * nc[nc > 40]
 380        lamc_neu = 0.951 * u.um / (1.0 + 3.616 / np.sqrt(m_ring))
 381        lamc_ion = 1.125 * u.um / (1.0 + 2.567 / np.sqrt(m_ring))
 382
 383        x = (1.0 * u.um / wavelength_arr).value
 384        c_abs_ion_out = np.zeros((len(radius_arr), len(wavelength_arr))) * u.cm**2
 385        c_abs_neu_out = np.zeros((len(radius_arr), len(wavelength_arr))) * u.cm**2
 386
 387        for i in range(len(radius_arr)):
 388            graph_cont = self.cabs_graphite_spl.ev(radius_arr[i].to(u.um), wavelength_arr.value) * u.cm**2
 389            c_abs_ion_out[i, :] += xi_gra[i] * graph_cont
 390            c_abs_neu_out[i, :] += xi_gra[i] * graph_cont
 391
 392            # PAH contribution
 393            c_fac_neu = c_func(
 394                (wavelength_arr / lamc_neu[i]).value ** -1
 395            )  # Note error in Equation A7, should be C(lambda_c/lambda)
 396            c_fac_ion = c_func(
 397                (wavelength_arr / lamc_ion[i]).value ** -1
 398            )  # Note error in Equation A7, should be C(lambda_c/lambda)
 399            s_mat_neu = np.zeros((len(self.lamj_tab), len(wavelength_arr))) * u.cm**2
 400            s_mat_ion = np.zeros((len(self.lamj_tab), len(wavelength_arr))) * u.cm**2
 401            for j in range(len(self.lamj_tab)):
 402                if self.hc_tab[j] == 0:
 403                    s_mat_neu[j, :] = s_func(wavelength_arr, self.lamj_tab[j], self.gamj_tab[j], self.sigj_neu_tab[j])
 404                    s_mat_ion[j, :] = s_func(wavelength_arr, self.lamj_tab[j], self.gamj_tab[j], self.sigj_ion_tab[j])
 405                else:
 406                    s_mat_neu[j, :] = s_func(
 407                        wavelength_arr, self.lamj_tab[j], self.gamj_tab[j], self.sigj_neu_tab[j] * hc[i]
 408                    )
 409                    s_mat_ion[j, :] = s_func(
 410                        wavelength_arr, self.lamj_tab[j], self.gamj_tab[j], self.sigj_ion_tab[j] * hc[i]
 411                    )
 412
 413            idx = np.where((10 < x) & (x < 15))
 414            c_abs_ion_out[i, idx] += (
 415                (s_mat_ion[0, idx] + (1.35 * x[idx] - 3.0) * 1.0e-18 * u.cm**2) * nc[i] * (1.0 - xi_gra[i])
 416            )
 417            c_abs_neu_out[i, idx] += (
 418                (s_mat_neu[0, idx] + (1.35 * x[idx] - 3.0) * 1.0e-18 * u.cm**2) * nc[i] * (1.0 - xi_gra[i])
 419            )
 420
 421            idx = np.where((7.7 < x) & (x <= 10))
 422            c_abs_ion_out[i, idx] += (
 423                (66.302 - 24.367 * x[idx] + 2.950 * x[idx] ** 2 - 0.1057 * x[idx] ** 3)
 424                * 1.0e-18
 425                * u.cm**2
 426                * nc[i]
 427                * (1.0 - xi_gra[i])
 428            )
 429            c_abs_neu_out[i, idx] += (
 430                (66.302 - 24.367 * x[idx] + 2.950 * x[idx] ** 2 - 0.1057 * x[idx] ** 3)
 431                * 1.0e-18
 432                * u.cm**2
 433                * nc[i]
 434                * (1.0 - xi_gra[i])
 435            )
 436
 437            idx = np.where((5.9 < x) & (x <= 7.7))
 438            c0 = 1.8687e-18 * u.cm**2
 439            c1 = 1.905e-19 * u.cm**2
 440            c2 = 4.175e-19 * u.cm**2
 441            c3 = 4.37e-20 * u.cm**2  # Note Equations A11 and A12 are mislabeled
 442            c_abs_ion_out[i, idx] += (
 443                (s_mat_ion[1, idx] + c0 + c1 * x[idx] + c2 * (x[idx] - 5.9) ** 2 + c3 * (x[idx] - 5.9) ** 3)
 444                * nc[i]
 445                * (1.0 - xi_gra[i])
 446            )
 447            c_abs_neu_out[i, idx] += (
 448                (s_mat_neu[1, idx] + c0 + c1 * x[idx] + c2 * (x[idx] - 5.9) ** 2 + c3 * (x[idx] - 5.9) ** 3)
 449                * nc[i]
 450                * (1.0 - xi_gra[i])
 451            )
 452
 453            idx = np.where((3.3 < x) & (x <= 5.9))
 454            c_abs_ion_out[i, idx] += (s_mat_ion[1, idx] + c0 + c1 * x[idx]) * nc[i] * (1.0 - xi_gra[i])
 455            c_abs_neu_out[i, idx] += (s_mat_neu[1, idx] + c0 + c1 * x[idx]) * nc[i] * (1.0 - xi_gra[i])
 456
 457            idx = np.where(x <= 3.3)
 458            c_abs_ion_out[i, idx] += (
 459                34.58e-18 * 10 ** (-3.431 / x[idx]) * u.cm**2 * c_fac_ion[idx] * nc[i] * (1.0 - xi_gra[i])
 460            )
 461            c_abs_neu_out[i, idx] += (
 462                34.58e-18 * 10 ** (-3.431 / x[idx]) * u.cm**2 * c_fac_neu[idx] * nc[i] * (1.0 - xi_gra[i])
 463            )
 464
 465            for j in range(len(self.lamj_tab)):
 466                if j > 1:
 467                    c_abs_ion_out[i, idx] += s_mat_ion[j, idx] * nc[i] * (1.0 - xi_gra[i])
 468                    c_abs_neu_out[i, idx] += s_mat_neu[j, idx] * nc[i] * (1.0 - xi_gra[i])
 469
 470            c_abs_ion_out[i, idx] += (
 471                3.5e-19 * 10 ** (-1.45 / x[idx]) * np.exp(-((0.1 * x[idx]) ** 2)) * nc[i] * (1.0 - xi_gra[i]) * u.cm**2
 472            )  # Note: does not appear in D21+
 473
 474        return c_abs_ion_out, c_abs_neu_out

 477        self,
 478        photon_wavelength_arr,
 479        dlambda,
 480        wavelength_arr,
 481        grain_radius,
 482        wavelength_arr_u,
 483        u_lambda_arr,
 484        p_lambda_arr,

 487        """Calculate the energy conservation normalization to scale basis vectors to the input radiation field.
 488
 489        Parameters
 490        ----------
 491        photon_wavelength_arr : astropy.units.Quantity (array_like)
 492            Array of absorbed photon wavelengths
 493        dlambda : float
 494            Width of the "monochromatic" radiation field, defined as a percentage of lambda_abs
 495        wavelength_arr : astropy.units.Quantity (array_like)
 496            Array of emission wavelengths
 497        grain_radius : astropy.units.Quantity (float)
 498            Dust grain radius
 499        wavelength_arr_u : astropy.units.Quantity (array_like)

 501        u_lambda_arr : astropy.units.Quantity (array_like)

 503        p_lambda_arr : astropy.units.Quantity (array_like)
 504            Basis vector array of size (len(photon_wavelength_arr), len(wavelength_arr)) for a given grain
 505        ion : bool
 506            Specifies whether or not the PAH is ionized, used to calculate the cross-section
 507
 508        Returns
 509        -------
 510        integrated_spectrum : astropy.units.Quantity (array_like)
 511            Integrated spectrum for the input grain and radiation field (in units of u.erg / (u.cm * u.s) with length
 512            len(wavelength_arr))
 513
 514        Raises
 515        ------
 516        AttributeError
 517            If the input is not an astropy.units.Quantity object
 518        TypeError
 519            If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 520        """
 521        wavelength_unit = u.um
 522        radiation_field_unit = u.erg / (u.cm**4)
 523        basis_vector_unit = u.erg / (u.cm * u.s)
 524        _check_param(photon_wavelength_arr, wavelength_unit, iterable=True)
 525        _check_param(photon_wavelength_arr, wavelength_unit, iterable=True)
 526        _check_param(wavelength_arr, wavelength_unit, iterable=True)
 527        _check_param(grain_radius, u.AA)
 528        _check_param(wavelength_arr_u, wavelength_unit, iterable=True)
 529        _check_param(u_lambda_arr, radiation_field_unit, iterable=True)
 530        _check_param(p_lambda_arr, basis_vector_unit, iterable=True)
 531
 532        # call calc_c_abs once for all wavelength values needed by calc_normalization
 533        lambda1_max = photon_wavelength_arr[-1] + photon_wavelength_arr[-1] * dlambda
 534        if ion:
 535            c_abs_arr_u = self.calc_c_abs(wavelength_arr_u, grain_radius)[0][0]
 536            c_abs_phot_arr = self.calc_c_abs(photon_wavelength_arr, grain_radius)[0][0]
 537            c_abs_phot_arr = np.insert(
 538                c_abs_phot_arr, len(c_abs_phot_arr), self.calc_c_abs(lambda1_max, grain_radius)[0][0]
 539            )
 540        else:
 541            c_abs_arr_u = self.calc_c_abs(wavelength_arr_u, grain_radius)[1][0]
 542            c_abs_phot_arr = self.calc_c_abs(photon_wavelength_arr, grain_radius)[1][0]
 543            c_abs_phot_arr = np.insert(
 544                c_abs_phot_arr, len(c_abs_phot_arr), self.calc_c_abs(lambda1_max, grain_radius)[1][0]
 545            )
 546
 547        normalizations = np.zeros(len(photon_wavelength_arr))
 548
 549        for i, lambda_abs in enumerate(photon_wavelength_arr):
 550            # define wavelengths of monochromatic radiation field (MRF)
 551            lambda0, lambda1 = lambda_abs, lambda_abs + lambda_abs * dlambda
 552            wh_mrf = np.where(np.logical_and(wavelength_arr_u > lambda0, wavelength_arr_u < lambda1))
 553            wavelength_arr_mrf = wavelength_arr_u[wh_mrf]
 554            wavelength_arr_mrf = np.insert(wavelength_arr_mrf, [0, len(wavelength_arr_mrf)], [lambda0, lambda1])
 555
 556            # get cross-section values of MRF
 557            c_abs_arr_mrf = c_abs_arr_u[wh_mrf]
 558            wh0, wh1 = np.argmin(np.abs(lambda0 - photon_wavelength_arr)), np.argmin(
 559                np.abs(lambda1 - photon_wavelength_arr)
 560            )
 561            c_abs_arr_mrf = np.insert(
 562                c_abs_arr_mrf, [0, len(c_abs_arr_mrf)], [c_abs_phot_arr[wh0], c_abs_phot_arr[wh1]]
 563            )
 564
 565            # interpolate radiation field to the MRF wavelength range
 566            u_lambda_arr_mrf = np.interp(wavelength_arr_mrf, wavelength_arr_u, u_lambda_arr)
 567
 568            # integrate to determine the power of the radiation field in this wavelength range
 569            numerator = trapezoid(u_lambda_arr_mrf * c_abs_arr_mrf * c.cgs, x=wavelength_arr_mrf.to(u.cm))
 570
 571            # integrate to determine the power radiated by the grain over all wavelengths
 572            denominator = trapezoid(p_lambda_arr[i], x=wavelength_arr.to(u.cm))
 573
 574            normalizations[i] = numerator / denominator
 575
 576        return p_lambda_arr * normalizations[:, np.newaxis]

 582def calc_pah_energy(grain_radius, temp_arr):
 583    """Calculate PAH vibrational energy as a function of temperature.
 584
 585    Parameters
 586    ----------
 587    grain_radius : astropy.units.Quantity (float)
 588        The PAH effective radius
 589    temp_arr : astropy.units.Quantity (array-like)
 590        Array of temperature to calculate energies for
 591
 592    Returns
 593    -------
 594    energy_arr : astropy.units.Quantity (array_like)
 595        Resulting PAH energy array (in u.erg)
 596
 597    Raises
 598    ------
 599    AttributeError
 600        If the input is not an astropy.units.Quantity object
 601    TypeError
 602        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 603    """
 604    radius_unit, temp_unit = u.AA, u.K
 605    _check_param(grain_radius, radius_unit)
 606    _check_param(temp_arr, temp_unit, iterable=True)
 607
 608    # 5 types of vibration: out-of-plane C-C modes, in-plane C-C modes, out-of-plane C-H bending, in-plane C-H bending,
 609    # and C-H stretching.
 610    nc = _calc_nc(grain_radius)  # number of carbon atoms
 611    nh = _calc_nh(nc)  # number of hydrogen atoms
 612    # TODO: consider turning these into functions
 613    nm_cc_op = nc - 2  # total number of C-C out-of-plane modes
 614    nm_cc_ip = 2 * (nc - 2)  # total number of C-C in-plane modes
 615
 616    # TODO: consider moving out of function and naming as constants __THETA_OP_CC
 617    # _THETA_OP_CC = 863 * u.K  # C-C out-of-plane bending mode Debye temperature
 618    # _THETA_IP_CC = 2500 * u.K  # C-C in-plane bending mode Debye temperature
 619    # _THETA_OP_CH = 1275 * u.K  # C-H out-of-plane beinding mode Debye temperature
 620    # _THETA_IP_CH = 1670 * u.K  # C-H in-plane bending mode Debye temperature
 621    # _THETA_STR_CH = 4360 * u.K  # C-H stretching mode Debye temperature
 622
 623    # determines if grain energy will be calculated using Debye spectrum method or discrete mode method
 624    nc_cutoff = 7360
 625
 626    energy_arr = None
 627
 628    # For grains with more carbon atoms than nc_cutoff, calculate C-C energies using the Debye spectrum approximation
 629    # Eq. 33 of Draine & Li (2001)
 630    if nc > nc_cutoff:
 631        energy_arr = _calc_pah_energy_debye(temp_arr, nh, nm_cc_ip, nm_cc_op)
 632
 633    # For grains with fewer carbon atoms than nc_cutoff, calculate energies by summing contributions from individual
 634    # modes
 635    # Eq. 2 of Draine & Li (2001)
 636    if nc <= nc_cutoff:
 637        energy_arr = _calc_pah_energy_modes(temp_arr, nc, nh, nm_cc_ip, nm_cc_op)
 638
 639    return energy_arr

 642def _calc_pah_energy_debye(temp_arr, nh, nm_cc_ip, nm_cc_op):
 643    """Calculate PAH energy using eq. 33 of Draine & Li (2001).
 644
 645    Parameters
 646    ----------
 647    temp_arr : astropy.units.Quantity (array-like)
 648        Array of temperature to calculate energies for
 649    nm_cc_ip : int
 650        Number of C-C in-plane stretching modes
 651    nm_cc_op : int
 652        Number of C-C out-of-plane stretching modes
 653    nh : int
 654        Number of hydrogen atoms
 655
 656    Returns
 657    -------
 658    energy_arr : astropy.units.Quantity (array_like)
 659        Resulting PAH energy array (in u.erg)
 660
 661    Raises
 662    ------
 663    AttributeError
 664        If the input is not an astropy.units.Quantity object
 665    TypeError
 666        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 667    """
 668    _check_param(temp_arr, u.K, iterable=True)
 669
 670    # contributions from individual C-H modes, as implemented in lines 157-181 of pah_spec_heat.f
 671    energy_ch = np.zeros(len(temp_arr)) * u.erg
 672
 673    x = _THETA_OP_CH / temp_arr
 674    y = exp(x)
 675    tmin = 32 * u.K
 676    energy_ch[temp_arr > tmin] += (
 677        nh * (k_B.cgs * temp_arr[temp_arr > tmin]) * (x[temp_arr > tmin] / (y[temp_arr > tmin] - 1))
 678    )
 679
 680    x = _THETA_IP_CH / temp_arr
 681    y = exp(x)
 682    tmin = 42 * u.K
 683    energy_ch[temp_arr > tmin] += (
 684        nh * (k_B.cgs * temp_arr[temp_arr > tmin]) * (x[temp_arr > tmin] / (y[temp_arr > tmin] - 1))
 685    )
 686
 687    x = _THETA_STR_CH / temp_arr
 688    y = exp(x)
 689    tmin = 109 * u.K
 690    energy_ch[temp_arr > tmin] += (
 691        nh * (k_B.cgs * temp_arr[temp_arr > tmin]) * (x[temp_arr > tmin] / (y[temp_arr > tmin] - 1))
 692    )
 693
 694    # contributions from C-C modes (approximated as Debye spectrum)
 695    energy_cc_op = np.zeros(len(temp_arr)) * u.erg
 696    energy_cc_op += _debye_2(_THETA_OP_CC / temp_arr) * k_B.cgs * temp_arr
 697
 698    energy_cc_ip = np.zeros(len(temp_arr)) * u.erg
 699    energy_cc_ip += _debye_2(_THETA_IP_CC / temp_arr) * k_B.cgs * temp_arr
 700
 701    # total energy
 702    energy_arr = energy_ch + nm_cc_op * energy_cc_op + nm_cc_ip * energy_cc_ip
 703
 704    return energy_arr

 707def _calc_pah_energy_modes(temp_arr, nc, nh, nm_cc_ip, nm_cc_op, return_modes=False):
 708    """Calculate PAH energy using eq. 2 of Draine & Li (2001).
 709
 710    Parameters
 711    ----------
 712    temp_arr : astropy.units.Quantity (array-like)
 713        Array of temperature to calculate energies for
 714    nc : int
 715        Number of carbon atoms
 716    nh : int
 717        Number of hydrogen atoms
 718    nm_cc_ip : int
 719        Number of C-C in-plane stretching modes
 720    nm_cc_op : int
 721        Number of C-C out-of-plane stretching modes
 722    return_modes : bool, optional
 723        Whether to return the array of pah mode energies
 724
 725    Returns
 726    -------
 727    energy_arr : astropy.units.Quantity (array_like)
 728        Resulting PAH energy array (in u.erg)
 729
 730    Raises
 731    ------
 732    AttributeError
 733        If the input is not an astropy.units.Quantity object
 734    TypeError
 735        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 736    """
 737    _check_param(temp_arr, u.K, iterable=True)
 738
 739    # TODO: consider creating a function specifically for the calculation of the vibrational mode energies and removing
 740    # the return_modes argument
 741    mode_arr_cc_op = _calc_cc_mode_energies(nc, nm_cc_op, _THETA_OP_CC)
 742    mode_arr_cc_ip = _calc_cc_mode_energies(nc, nm_cc_ip, _THETA_IP_CC)
 743    mode_arr_ch_op = _calc_ch_mode_energies(nh, _THETA_OP_CH)
 744    mode_arr_ch_ip = _calc_ch_mode_energies(nh, _THETA_IP_CH)
 745    mode_arr_ch_str = _calc_ch_mode_energies(nh, _THETA_STR_CH)
 746
 747    emode_arr = (
 748        sorted(
 749            np.concatenate(
 750                (
 751                    mode_arr_cc_op.value,
 752                    mode_arr_cc_ip.value,
 753                    mode_arr_ch_op.value,
 754                    mode_arr_ch_ip.value,
 755                    mode_arr_ch_str.value,
 756                )
 757            )
 758        )
 759        * mode_arr_cc_op.unit
 760    )
 761    nmodes = len(emode_arr)
 762
 763    energy_arr = np.zeros(len(temp_arr)) * u.erg
 764
 765    exp_cutoff = 100  # ignore contributions from exp(x) when x is large
 766
 767    # emodes array contains mode energies from C-C in-plane and out-of-plane bending and C-H stretching and in-plane
 768    # in-plane and out-of-plane bending
 769    for j in range(0, nmodes):
 770        x = emode_arr[j] / (k_B.cgs * temp_arr)
 771        y = exp(x)
 772        energy_arr[x < exp_cutoff] += (x[x < exp_cutoff] / (y[x < exp_cutoff] - 1)) * k_B.cgs * temp_arr[x < exp_cutoff]
 773
 774    if return_modes:
 775        return energy_arr, emode_arr
 776
 777    return energy_arr

 780def _calc_pah_cooling(lambda_abs, grain_radius, wavelength_arr, c_abs_arr, temp_arr, energy_arr):
 781    """Calculate the temperature evolution of a PAH following a single-photon absorption.
 782
 783    Parameters
 784    ----------
 785    lambda_abs : astropy.units.Quantity (float)
 786        Wavelength of absorbed photon
 787    grain_radius : astropy.units.Quantity (float)
 788        Dust grain radius
 789    wavelength_arr : astropy.units.Quantity (array_like)
 790        Emission wavelengths to integrate over
 791    c_abs_arr : astropy.units.Quantity (array_like)
 792        Array of length len(wavelength_arr) with C_abs values for a grain of size grain_radius
 793    temp_arr : astropy.units.Quantity (array_like)
 794        Array of temperatures corresponding to PAH vibrational energies for a grain of size grain_radius
 795    energy_arr : astropy.units.Quantity (array_like)
 796        Array of PAH vibrational energies as a function of temperature for a grain of size grain_radius
 797
 798    Returns
 799    -------
 800    dt_arr_out : astropy.units.Quantity (array_like)
 801        Array of time-steps used to solve for T(t) (in u.s)
 802    time_arr_out : astropy.units.Quantity (array_like)
 803        Array with time values for T(t) (in u.s)
 804    temp_arr_out : astropy.units.Quantity (array_like)
 805        Array with temperature values for T(t) (in u.K)
 806
 807    Raises
 808    ------
 809    AttributeError
 810        If the input is not an astropy.units.Quantity object
 811    TypeError
 812        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 813    """
 814    wavelength_unit, radius_unit, c_abs_unit, energy_unit, temp_unit = u.um, u.AA, u.cm**2, u.erg, u.K
 815    _check_param(lambda_abs, wavelength_unit)
 816    _check_param(grain_radius, radius_unit)
 817    _check_param(wavelength_arr, wavelength_unit, iterable=True)
 818    _check_param(c_abs_arr, c_abs_unit, iterable=True)
 819    _check_param(energy_arr, energy_unit, iterable=True)
 820    _check_param(temp_arr, temp_unit, iterable=True)
 821
 822    nu_arr = c.cgs / wavelength_arr.to(u.cm)
 823
 824    energy_abs = c.cgs * h.cgs / lambda_abs.to(u.cm)
 825    temp_abs = np.interp(energy_abs.value, energy_arr.value, temp_arr.value) * u.K
 826
 827    print(f"Photon wavelength: {lambda_abs:.2f}, initial temperature: {temp_abs:.2f}")
 828
 829    time_i, energy_i, temp_i = 0 * u.s, energy_abs, temp_abs
 830    time_arr_out, temp_arr_out, dt_arr_out = [0], [temp_i.value], []
 831    dt_unit, time_unit, temp_unit = None, None, None
 832    # The change in energy per timestep, results are converged for dE < 3%
 833    dE_max = 0.03
 834
 835    while temp_i.value > 5.1:
 836        dE_dt = -trapezoid(4 * np.pi * _planck_function_nu(nu_arr, temp_i) * c_abs_arr, x=nu_arr)
 837
 838        dt = dE_max * energy_i / dE_dt
 839        dE = dE_dt * dt
 840
 841        energy_i -= dE
 842        time_i += dt
 843
 844        temp_i = np.interp(energy_i.value, energy_arr.value, temp_arr.value) * u.K
 845
 846        dt_arr_out.append(dt.value)
 847        time_arr_out.append(time_i.value)
 848        temp_arr_out.append(temp_i.value)
 849        dt_unit, time_unit, temp_unit = dt.unit, time_i.unit, temp_i.unit
 850
 851    dt_arr_out = np.array(dt_arr_out) * dt_unit
 852    time_arr_out = np.array(time_arr_out) * time_unit
 853    temp_arr_out = np.array(temp_arr_out) * temp_unit
 854
 855    print(f"Final temperature of {temp_arr_out[-1]:.2f} at time {time_arr_out[-1]:.2e}")
 856
 857    return dt_arr_out, time_arr_out, temp_arr_out

 862):

 868        Wavelength of absorbed photon

 874        Array of length len(wavelength_arr) with C_abs values for a grain of size grain_radius
 875    temp_arr : astropy.units.Quantity (array_like)

 878        Array of PAH vibrational energies as a function of temperature for a grain of size grain_radius

 885        Array of time-steps used to solve for T(t) (in u.s)
 886    time_arr_out : astropy.units.Quantity (array_like)

 890
 891    Raises
 892    ------
 893    AttributeError
 894        If the input is not an astropy.units.Quantity object
 895    TypeError
 896        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
 897    """
 898    wavelength_unit, radius_unit, c_abs_unit, energy_unit, temp_unit = u.um, u.AA, u.cm**2, u.erg, u.K
 899    _check_param(lambda_abs, wavelength_unit)
 900    _check_param(grain_radius, radius_unit)
 901    _check_param(wavelength_arr, wavelength_unit, iterable=True)
 902    _check_param(c_abs_arr, c_abs_unit, iterable=True)
 903    _check_param(energy_arr, energy_unit, iterable=True)
 904    _check_param(temp_arr, temp_unit, iterable=True)
 905
 906    nu_arr = c.cgs / wavelength_arr.to(u.cm)
 907
 908    energy_abs = c.cgs * h.cgs / lambda_abs.to(u.cm)
 909    temp_abs = np.interp(energy_abs.value, energy_arr.value, temp_arr.value) * u.K
 910
 911    print(f"Photon wavelength: {lambda_abs:.2f}, initial temperature: {temp_abs:.2f}")
 912
 913    time_i, energy_i, temp_i = 0 * u.s, energy_abs, temp_abs
 914    time_arr_out, temp_arr_out, dt_arr_out = [0], [temp_i.value], []
 915    dt_unit, time_unit, temp_unit = None, None, None
 916    # The change in energy per timestep, results are converged for dE < 3%
 917    dE_max = 0.03
 918
 919    # Define first 10 energy bins following method in DL01 Appendix B
 920    nbins = 10
 921
 922    energy_j_min = np.zeros(nbins) * u.erg
 923    energy_j_max = np.zeros(nbins) * u.erg
 924
 925    # Eq. B1
 926    energy_j_min[1] = 3 / 2 * pah_energy_modes[1] - 1 / 2 * pah_energy_modes[2]
 927    energy_j_max[0] = energy_j_min[1]
 928
 929    # Eq. B2
 930    for j in range(0, 2):
 931        bin_edge = 1 / 2 * (pah_energy_modes[j] + pah_energy_modes[j + 1])
 932        energy_j_max[j] = bin_edge
 933        energy_j_min[j + 1] = bin_edge
 934
 935    # Eq. B3
 936    for j in range(2, nbins):
 937        bin_edge = 1 / 2 * (pah_energy_modes[2 * j - 1] + pah_energy_modes[2 * j])
 938        energy_j_max[j] = bin_edge
 939        if j != nbins - 1:
 940            energy_j_min[j + 1] = bin_edge
 941
 942    # exclude emission from frequencies exceeding the frequency of the absorbed photon
 943    weights_sawteeth = np.ones(len(nu_arr))
 944    weights_sawteeth[nu_arr > c.cgs / lambda_abs.to(u.cm)] = 0
 945
 946    # Cache variables outside of loop to modify the final temperature bin before manually implementing temperature bins
 947    dE, dt, dE_dt = None, None, None
 948    while energy_i > energy_j_max[-1]:
 949        dE_dt = -trapezoid(4 * np.pi * _planck_function_nu(nu_arr, temp_i) * c_abs_arr * weights_sawteeth, x=nu_arr)
 950
 951        dt = dE_max * energy_i / dE_dt
 952        dE = dE_dt * dt
 953
 954        energy_i -= dE
 955        time_i += dt
 956
 957        temp_i = np.interp(energy_i.value, energy_arr.value, temp_arr.value) * u.K
 958
 959        dt_arr_out.append(dt.value)
 960        time_arr_out.append(time_i.value)
 961        temp_arr_out.append(temp_i.value)
 962        dt_unit, time_unit, temp_unit = dt.unit, time_i.unit, temp_i.unit
 963
 964    # Ensure that the final temperature bin lines up with the pre-defined energy bins
 965    # Undo the final timestep
 966    energy_i += dE
 967    time_i -= dt
 968
 969    # Calculate the intended timestep
 970    dE_correction = energy_i - energy_j_max[-1]
 971    dt_correction = dE_correction / dE_dt
 972
 973    # Apply the correction
 974    energy_i -= dE_correction  # should line up with energy_j_max[-1]
 975    time_i += dt_correction
 976    temp_i = np.interp(energy_i, energy_arr, temp_arr)
 977
 978    # Overwrite the final values
 979    dt_arr_out[-1] = dt_correction.value
 980    time_arr_out[-1] = time_i.value
 981    temp_arr_out[-1] = temp_i.value
 982
 983    for j in range(nbins - 1, -1, -1):
 984        dE_dt = -trapezoid(4 * np.pi * _planck_function_nu(nu_arr, temp_i) * c_abs_arr * weights_sawteeth, x=nu_arr)
 985
 986        dE = energy_j_max[j] - energy_j_min[j]
 987        dt = dE / dE_dt
 988
 989        energy_i -= dE
 990        time_i += dt
 991        temp_i = np.interp(energy_i.value, energy_arr.value, temp_arr.value) * u.K
 992
 993        dt_arr_out.append(dt.value)
 994        time_arr_out.append(time_i.value)
 995        temp_arr_out.append(temp_i.value)
 996
 997        dt_unit, time_unit, temp_unit = dt.unit, time_i.unit, temp_i.unit
 998
 999    # ensure that zero energy corresponds to zero temperature (np.interp sometimes returns nonzero/NaN values)
1000    temp_arr_out[-1] = 0
1001

1011def _calc_basis_vector(grain_radius, lambda_abs, wavelength_arr, c_abs_arr, weighting_arr, temp_arr):
1012    """Calculate the basis vector for a single-photon absorption.
1013
1014    Parameters
1015    ----------
1016    grain_radius : astropy.units.Quantity (float)
1017        Dust grain radius
1018    lambda_abs : astropy.units.Quantity (float)
1019        Wavelength of absorbed photon
1020    wavelength_arr : astropy.units.Quantity (array_like)
1021        Array of emission wavelengths
1022    c_abs_arr : astropy.units.Quantity (array_like)
1023        Array of length len(wavelength_arr) with C_abs values for a given grain
1024    weighting_arr : array_like
1025        Array of length len(temp_arr) with values to weight temperatures by
1026    temp_arr : astropy.units.Quantity (array_like)
1027        Array with temperature values for T(t) (upper energy bin edges only, in u.K)
1028
1029    Returns
1030    -------
1031    basis_vector : astropy.units.Quantity (array_like)
1032        Array of floats of length len(wavelength_arr) with basis vector for a given grain (in u.erg / (u.cm * u.s))
1033
1034    Raises
1035    ------
1036    AttributeError
1037        If the input is not an astropy.units.Quantity object
1038    TypeError
1039        If the astropy.units.Quantity object has incorrect units (or optionally is not array-like)
1040    """
1041    wavelength_unit, temp_unit, c_abs_unit = u.um, u.K, u.cm**2
1042    _check_param(lambda_abs, wavelength_unit)
1043    _check_param(wavelength_arr, wavelength_unit, iterable=True)
1044    _check_param(temp_arr, temp_unit, iterable=True)
1045    _check_param(c_abs_arr, c_abs_unit, iterable=True)
1046    _check_param(weighting_arr, u.dimensionless_unscaled)
1047
1048    unit = None
1049
1050    weighting_energy_arr = calc_pah_energy(grain_radius=grain_radius, temp_arr=temp_arr)
1051
1052    basis_vector = np.zeros(len(wavelength_arr))
1053    for i, lambda_i in enumerate(wavelength_arr.to(u.cm)):
1054        weights_sawteeth = np.ones(len(weighting_energy_arr))
1055        # Condition in Eq. 56 of DL01 excluding emission from lambda_em such that E(lambda_em) > E(lambda_abs)
1056        weights_sawteeth[h.cgs * c.cgs / lambda_i.to(u.cm) > weighting_energy_arr] = 0
1057        p_lambda_i = np.sum(
1058            4
1059            * np.pi
1060            * _planck_function_lambd(lambda_i.to(u.cm), temp_arr)
1061            * weights_sawteeth
1062            * weighting_arr
1063            * c_abs_arr[i]
1064        )
1065        unit = p_lambda_i.unit
1066        basis_vector[i] = p_lambda_i.value
1067
1068    return basis_vector * unit

372            denom = np.pi * (((lam / lamj) - (lamj / lam)) ** 2 + gamma**2)
373            return num / denom

376            return np.arctan(1.0e3 * (y - 1.0) ** 3 / y) / np.pi + 0.5

440            c1 = 1.905e-19 * u.cm**2
